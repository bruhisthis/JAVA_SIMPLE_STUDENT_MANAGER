public class CorrGradeDAOImpl implements GradeDAO {
    private static final Logger logger = AppLogger.getLogger(GradeDAOImpl.class);
    
    // ========== VALIDATION METHODS ==========
    
    private void validateId(Integer id) throws InvalidInputException {
        if (id == null || id <= 0) {
            throw new InvalidInputException("Invalid input",null,id,null,null);
        }
    }
    
    private void validateScore(Double score) throws InvalidInputException {
        if (score == null || score < 0 || score > 100) {
            throw new InvalidInputException("Invalid input : " ,null,null,score,INPUTYPE.DOUBLE);
        }
    }
    
    private void validateSubject(String subject) throws InvalidInputException {
        if (subject == null || subject.isBlank()) {
            throw new InvalidInputException("inavalid input: ",subject,null,null,INPUTYPE.STRING);
        }
    }
    
    private void validateGrade(Grade grade) throws InvalidInputException {
        if (grade == null) {
            throw new InvalidInputException();
        }
        validateId(grade.getStudentId());
        validateScore(grade.getScore());
        validateSubject(grade.getSubject());
    }
    
    // ========== HELPER METHOD ==========
    
    private Grade extractGradeFromResultSet(ResultSet rs) throws SQLException {
        int id = rs.getInt("id");
        int studentId = rs.getInt("student_id"); // ✅ Correct column name
        String subject = rs.getString("subject");
        double score = rs.getDouble("score");
        Timestamp timestamp = rs.getTimestamp("exam_date");
        
        Grade grade = new Grade(id, studentId, subject, score);
        if (timestamp != null) {
            grade.setExamDate(timestamp.toLocalDateTime());
        }
        
        return grade;
    }
    
    // ========== CRUD OPERATIONS ==========
    
    @Override
    public Integer addGrade(Grade grade) throws DAOException, BaseExceptionApp {
        validateGrade(grade);
        
        String sql = "INSERT INTO grades (student_id, subject, score, exam_date) VALUES (?, ?, ?, ?)";
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            
            pstmt.setInt(1, grade.getStudentId());
            pstmt.setString(2, grade.getSubject());
            pstmt.setDouble(3, grade.getScore());
            
            LocalDateTime examDate = grade.getExamDate();
            if (examDate != null) {
                pstmt.setTimestamp(4, Timestamp.valueOf(examDate));
            } else {
                pstmt.setTimestamp(4, null);
            }
            
            int rows = pstmt.executeUpdate();
            if (rows == 0) {
                throw new DAOException("Insert failed: no rows affected", null);
            }
            
            try (ResultSet rs = pstmt.getGeneratedKeys()) {
                if (rs.next()) {
                    int id = rs.getInt(1);
                    logger.info("Grade inserted for student " + grade.getStudentId() + 
                               " in " + grade.getSubject() + " (ID: " + id + ")");
                    return id;
                }
            }
            
            throw new DAOException("Insert succeeded but ID not returned", null);
            
        } catch (SQLException e) {
            // ✅ Handle Foreign Key violation
            if (e.getSQLState() != null && e.getSQLState().equals("23503")) {
                logger.log(Level.WARNING, "Student not found: " + grade.getStudentId(), e);
                throw new InvalidInputException("invalid input : ",null,grade.getStudentId(),null,null);
            }
            logger.log(Level.SEVERE, "Failed to insert grade", e);
            throw new DAOException("Database insertion failed", e);
        }
    }
    
    @Override
    public void updatescore(Integer gradeId, Double score) throws DAOException, BaseExceptionApp {
        validateId(gradeId);
        validateScore(score);
        
        String sql = "UPDATE grades SET score = ? WHERE id = ?";
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            
            pstmt.setDouble(1, score);
            pstmt.setInt(2, gradeId);
            
            int rows = pstmt.executeUpdate();
            if (rows == 0) {
                logger.warning("Update failed: grade not found with ID " + gradeId);
                throw new DAOException("Grade not found: " + gradeId, null);
            }
            
            logger.info("Score updated for grade ID " + gradeId + ": " + score);
            
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Failed to update score for grade ID " + gradeId, e);
            throw new DAOException("Database update failed", e);
        }
    }
    
    @Override
    public void updatesExamDate(Integer gradeId, LocalDateTime examDate) throws DAOException, BaseExceptionApp {
        validateId(gradeId);
        if (examDate == null) {
            throw new InvalidInputException("Exam date cannot be null",null,null,null,null);
        }
        
        String sql = "UPDATE grades SET exam_date = ? WHERE id = ?";
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            
            pstmt.setTimestamp(1, Timestamp.valueOf(examDate));
            pstmt.setInt(2, gradeId);
            
            int rows = pstmt.executeUpdate();
            if (rows == 0) {
                logger.warning("Update failed: grade not found with ID " + gradeId);
                throw new DAOException("Grade not found: " + gradeId, null);
            }
            
            logger.info("Exam date updated for grade ID " + gradeId);
            
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Failed to update exam date for grade ID " + gradeId, e);
            throw new DAOException("Database update failed", e);
        }
    }
    
    @Override
    public void hardDeleteGrade(Integer id) throws DAOException, BaseExceptionApp {
        validateId(id);
        
        String sql = "DELETE FROM grades WHERE id = ?";
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            
            pstmt.setInt(1, id);
            
            int rows = pstmt.executeUpdate();
            if (rows == 0) {
                logger.warning("Delete failed: grade not found with ID " + id);
                throw new DAOException("Grade not found: " + id, null);
            }
            
            logger.info("Grade deleted: ID " + id);
            
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Failed to delete grade ID " + id, e);
            throw new DAOException("Database delete failed", e);
        }
    }
    
    @Override
    public Optional<GradeBase> findGradeSById(Integer id) throws DAOException, BaseExceptionApp {
        validateId(id);
        
        String sql = "SELECT * FROM grades WHERE id = ?";
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql)) {
            
            pstmt.setInt(1, id);
            
            try (ResultSet rs = pstmt.executeQuery()) {
                if (rs.next()) {
                    Grade grade = extractGradeFromResultSet(rs);
                    logger.info("Grade found: ID " + id);
                    return Optional.of(grade);
                }
                
                logger.info("Grade not found: ID " + id);
                return Optional.empty();
            }
            
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Failed to find grade by ID " + id, e);
            throw new DAOException("Database query failed", e);
        }
    }
    
    @Override
    public List<Grade> findAllGrade() throws DAOException {
        String sql = "SELECT * FROM grades ORDER BY student_id, subject";
        List<Grade> grades = new ArrayList<>();
        
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement pstmt = conn.prepareStatement(sql);
             ResultSet rs = pstmt.executeQuery()) {
            
            while (rs.next()) {
                grades.add(extractGradeFromResultSet(rs));
            }
            
            logger.info("Retrieved " + grades.size() + " grades");
            return grades;
            
        } catch (SQLException e) {
            logger.log(Level.SEVERE, "Failed to retrieve all grades", e);
            throw new DAOException("Database query failed", e);
        }
    }
